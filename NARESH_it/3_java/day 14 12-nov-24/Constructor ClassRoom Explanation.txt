{
	int l;
	int b;
	
	Room() {
		System.out.println("NPC1");
	}
	Room() {
		System.out.println("NPC2");
	}
	public void setL(int l) {
		this.l = l;
	}
	public void setB(int b) {
		this.b = b;
	}
	
	public int getL() {
		return l;
	}
	public int getB() {
		return b;
	}

		System.out.println("Area is : "+(l*b));
