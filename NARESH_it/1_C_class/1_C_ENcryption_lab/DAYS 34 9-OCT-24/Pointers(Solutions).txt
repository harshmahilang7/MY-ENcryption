#include<stdio.h>
void mergeAndSort(int *a, int l1, int *b, int l2, int *c);
double findMedian(int *c, int l3);
main()
{
	int a[]={1,2,3};
	int b[]={5,6,7,8};
	int l1=sizeof(a)/sizeof(int);
	int l2=sizeof(b)/sizeof(int);
	int c[l1+l2];
	mergeAndSort(a,l1,b,l2,c);
	double res= findMedian(c,l1+l2);
	printf("%.1f",res);
	
	
}

void mergeAndSort(int *a, int l1, int *b, int l2, int *c)
{
	int x=0;
	for(int i=0;i<l1;i++)
	{
		*(c+x++)=*(a+i);
	}
	
	for(int i=0;i<l2;i++)
	{
		*(c+x++)=*(b+i);
	}
	
	for(int i=0;i<l1+l2;i++)
	{
		for(int j=i+1;j<l1+l2;j++)
		{
			if(*(c+i)>*(c+j)) 
			{
				int t=*(c+i);
				*(c+i)=*(c+j);
				*(c+j)=t;
			}
		}
	}
}

double findMedian(int *c, int l3) 
{
	if(l3%2!=0)
	{
		return *(c+(l3/2));
	}
	else
	{
	   return (*(c+l3/2)+ *(c+l3/2-1))/2.0;
	}
}

#include<stdio.h>
void rotateRight(int *a, int l, int p);
main()
{
	int a[]={1,2,3,4,5};
	int l=sizeof(a)/sizeof(int);
	rotateRight(a,l,3);	
	for(int i=0;i<l;i++)
	{
		printf("%d ",a[i]);
	}
}

void rotateRight(int *a, int l, int p)
{	 
	 	for(int k=1;k<=p;k++)
	 	{
	 		int t= *(a+l-1); 
		 for(int i=l-1;i>=0;i--)
		 {
		 	*(a+i)=*(a+i-1);
		 }
		  
		 *(a+0)=t;
		}
}

#include<stdio.h>
#include<string.h>
void removeSp(char *s,int l);
int isPalindrom(char *s,int l);
main()
{
    char s[]="A man,a plan, a canal: Panama";
    int l=strlen(s);
    removeSp(s,l);
    l=strlen(s);
    if(isPalindrom(s,l))
    printf("Palindrom");
    else
    printf("Not Palindrom");
    
    
}

int isPalindrom(char *s,int l)
{
	 for(int i=0;i<l;i++)
	 {
	 	if(*(s+i)!=*(s+l-1))
	 	{
	 		printf("%c != %c\n",s[i],s[l-1]);
	 		return 0;
		 }	
		 --l;
	 }
	 return 1;
	 
}

void removeSp(char *s,int l)
{	 
	 for(int i=0;i<l;i++)
	 {
	    if(*(s+i)>='A' && *(s+i)<='Z')
		{
		  *(s+i)=*(s+i)+32;	
		}	
	 }
	 
	 for(int i=0;i<l;i++)
	 {
	 	if(!(*(s+i)>='a' && *(s+i)<='z'))
	 	{
	 		for(int j=i;j<l;j++)
	 		{
	 			*(s+j)=*(s+j+1);
			 }
			 i--;
			 l--;
		 }
	 }	
}
